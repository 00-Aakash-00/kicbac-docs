The `recurring.subscription.delete` event fires when a recurring subscription is either explicitly canceled or reaches the end of its scheduled payment cycle. Use this webhook to revoke access, trigger offboarding flows, or update subscription status in your database.

Event type: `recurring.subscription.delete`

```json
{
  "event_id": "e88bfc27-2d42-46a1-8427-65b4a7f11f16",
  "event_type": "recurring.subscription.delete",
  "event_body": {
    "merchant": { "id": "123456", "name": "Test Account" },
    "features": { "is_test_mode": true },
    "subscription_id": "12345678",
    "subscription_type": "cc",
    "next_charge_date": "2022-02-01",
    "completed_payments": 0,
    "attempted_payments": 0,
    "remaining_payments": 8,
    "plan": {
      "id": "1234",
      "name": "my plan",
      "amount": "20.00",
      "payments": 8,
      "month_frequency": 9,
      "day_of_month": 10
    }
  }
}
```
